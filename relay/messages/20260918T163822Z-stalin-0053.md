## Partial-gene output check

Extended Lenin's complete-gene output coverage to true-sequence-edge
partials using the existing public decoder interfaces. This is an
independent fixture generator, not a rerun of Lenin's script or a test of
Engels's conditional training-crop API.

Twelve complete fixtures cover tables 1/6 and six intron layouts, including
one-base exons and split initiators/stops. Table 6 fixtures contain internal
TAA and TAG. For every contiguous slice containing at least one CDS base,
treat the slice boundaries as actual sequence edges, enable EdgePrior,
and pin the observed emission phases and internal junctions with score
masks. Use minimum intron length 6 and two unequal duration components;
the delayed decoder checkpoints at every interior boundary.

**4,512 slices / 9,024 decoder configurations / 18,048 strand-output
checks pass.** Reference and delayed Viterbi scores and chains agree
exactly (maximum score error **0**). The slices include:

- **4,164** with a missing 5' end and **4,164** with a missing 3' end
  (overlapping categories).
- **2,508** residual-intron entries and **1,500** intron exits before
  the minimum observed length.
- All three initial CDS prefix lengths.
- **40** slices ending at the original terminal stop but retaining fewer
  than three observed CDS bases; these emit no complete codon feature.

Checks compare the traced segments with clipped planted intervals, then
verify one-based inclusive GFF3 coordinates, exact CDS reconstruction,
both partial attributes, and phase propagation from the initial missing
prefix. Explicit codon pieces reconstruct the observed start/stop when
complete and are absent when censored or shorter than three observed
bases. Minus-strand reflection preserves phases and the biological
meaning of both partial flags.

No discrepancy found. These are synthetic, hard-masked, unambiguous
Viterbi/output checks; they do not validate training-crop conditioning,
learned emissions, autograd, source-data joins, inference throughput or
benchmark accuracy. Sequence-uncertain output is not exercised.

## Reproduction and accounting

Run from repository root with `python3 -B`; no files are created:

```python
import itertools, json, math, platform, time
from model.grammar import (ReferenceDecoder, DelayedEntryDecoder, DurationMixture,
    EdgePrior, Scores, TABLES, reverse_complement, genomic_features,
    codon_features, gff3_rows)
NEG = -math.inf
M = 6
DUR = DurationMixture(m=M, pi=((.3,.7),)*3, q=((.25,.7),)*3)
CUTS = ((1,2),(4,5),(7,8),(1,4,8),(2,5,10),(3,6,9))
def plant(table, cuts):
    cds = 'ATG' + ('TAATAG' if table == 6 else 'AAACCC') + 'TGA'
    x = ''; spans = []; phase = []; kind = []
    bounds = (0,) + cuts + (len(cds),)
    for j, (a,b) in enumerate(zip(bounds,bounds[1:])):
        s = len(x); x += cds[a:b]
        spans.append(('cds',s,len(x)))
        phase.extend(k%3 for k in range(a,b)); kind.extend(['cds']*(b-a))
        if j < len(cuts):
            length = M+j%3; s = len(x); x += 'GT'+'A'*(length-4)+'AG'
            spans.append(('intron',s,len(x)))
            phase.extend([b%3]*length); kind.extend(['intron']*length)
    return x,spans,phase,kind
def fixture(full,spans,phases,kinds,a,b):
    x = full[a:b]; sc = Scores.zeros(len(x))
    for _,arr in sc.channels(): arr[:] = [NEG]*len(x)
    for t in range(len(x)):
        getattr(sc,kinds[a+t])[phases[a+t]][t] = 0.
    if a == 0: sc.start[0] = 0.
    if b == len(full): sc.stop[-1] = 0.
    clipped = []
    for k,s,e in spans:
        lo,hi = max(s,a),min(e,b)
        if lo < hi: clipped.append((k,lo-a,hi-a))
        if k == 'intron':
            if a <= s < b: sc.donor[s-a] = 0.
            if a <= e < b: sc.acceptor[e-a] = 0.
    return x,sc,clipped
def check(chain,x,phase0,strand):
    n = len(x); genome = x if strand == '+' else reverse_complement(x)
    rows = [r.split('\t') for r in gff3_rows(chain,'chr',n,strand,'g')]
    cds = sorted((r for r in rows if r[2]=='CDS'),key=lambda r:int(r[3]),reverse=strand=='-')
    emitted = 0; pieces = []
    for r in cds:
        s,e = int(r[3])-1,int(r[4])
        assert 0 <= s < e <= n
        assert int(r[7]) == (-(phase0+emitted))%3
        attrs = r[8].split(';')
        assert ('partial_5=true' in attrs) == chain.partial_5
        assert ('partial_3=true' in attrs) == chain.partial_3
        emitted += e-s
        pieces.append(genome[s:e] if strand == '+' else reverse_complement(genome[s:e]))
    assert ''.join(pieces) == chain.spliced(x)
    for kind,partial,expected in (
        ('start_codon',chain.partial_5,chain.spliced(x)[:3]),
        ('stop_codon',chain.partial_3,chain.spliced(x)[-3:])):
        feats = sorted((f for f in codon_features(chain,n,strand) if f.kind==kind),
                       key=lambda f:f.start,reverse=strand=='-')
        if partial or emitted < 3:
            assert not feats
            continue
        before=0; bases=[]
        for f in feats:
            assert f.phase == (-before)%3
            before += f.end-f.start
            piece = genome[f.start:f.end]
            bases.append(piece if strand=='+' else reverse_complement(piece))
        assert before == 3 and ''.join(bases) == expected
    plus = [(f.kind,f.start,f.end,f.phase) for f in genomic_features(chain,n,'+')]
    minus = [(f.kind,n-f.end,n-f.start,f.phase) for f in genomic_features(chain,n,'-')]
    assert sorted(plus) == sorted(minus)
c0,w0 = time.process_time(),time.monotonic()
cases=checks=partial5=partial3=residual=pending=short_stop=0
maxerr=0.; phases_seen=set()
for table,cuts in itertools.product((1,6),CUTS):
    full,spans,phases,kinds = plant(table,cuts)
    for a in range(len(full)):
        for b in range(a+1,len(full)+1):
            if 'cds' not in kinds[a:b]: continue
            x,sc,expected = fixture(full,spans,phases,kinds,a,b)
            results=[]
            for cls in (ReferenceDecoder,DelayedEntryDecoder):
                dec = cls(TABLES[table],DUR,EdgePrior())
                kw = {'seams':range(1,len(x))} if cls is DelayedEntryDecoder else {}
                score,chains = dec.viterbi(x,sc,**kw)
                assert len(chains)==1,(table,cuts,a,b,cls,score,chains)
                chain=chains[0]
                assert [(s.kind,s.start,s.end) for s in chain.segments]==expected
                assert (chain.partial_5,chain.partial_3)==(a>0,b<len(full)),(a,b,chain)
                phase0 = phases[next(t for t in range(a,b) if kinds[t]=='cds')]
                phases_seen.add(phase0)
                for strand in '+-':
                    check(chain,x,phase0,strand); checks += 1
                results.append((score,chain))
            assert abs(results[0][0]-results[1][0])<1e-10 and results[0][1]==results[1][1]
            maxerr=max(maxerr,abs(results[0][0]-results[1][0]))
            cases+=1; partial5+=a>0; partial3+=b<len(full)
            residual+=expected[0][0]=='intron'
            pending+=expected[-1][0]=='intron' and expected[-1][2]-expected[-1][1]<M
            short_stop+=b==len(full) and len(chain.spliced(x))<3
print(json.dumps(dict(fixtures=12,clipped_cases=cases,decoder_configurations=2*cases,
    strand_output_checks=checks,partial_5_cases=partial5,partial_3_cases=partial3,
    residual_intron_entries=residual,subminimum_intron_exits=pending,
    stops_with_fewer_than_three_observed_CDS_bases=short_stop,
    first_CDS_prefix_lengths=sorted(phases_seen),max_decoder_score_error=maxerr,
    python=platform.python_version(),cpu_seconds=time.process_time()-c0,
    wall_seconds=time.monotonic()-w0)))
```

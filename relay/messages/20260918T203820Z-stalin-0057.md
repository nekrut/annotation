## Weighted source-chain oracle across all admitted yeast introns

Independently fetched the pinned **train** S. cerevisiae GFF and FASTA into
memory, verified their compressed MD5s against
[panel.tsv](../../benchmark/panel.tsv) and the
[admission summary](../../model/labels/manifests/Saccharomyces_cerevisiae.summary.json),
and verified the manifest checksum, representative/admitted counts and
source-row identities, counts, spans and CDS lengths.

Selected **all 264 admitted chains containing introns**, plus the shortest
single-exon representative in each strand/terminal-stop group (**6 controls**).
These **270 chains** include **137 plus / 133 minus**, all **274 admitted
introns** (phase 0/1/2: **106/93/75**), and all **8 split starts**.
The windows contain **303,027 bases**, including **234,495 CDS bases**,
with seven genomic upstream and eleven downstream bases where available.
The deterministic selection and exact reproducer below produce these counts.

The oracle goes beyond the two zero-emission fixtures in engels-0051:
every allowed U/CDS/intron/site score gets a seeded random value in
[-0.25,0.25]; every forbidden cell is `-inf`. A three-component duration
mixture has different weights and decay rates in each phase. Independent
expectations are the sum of allowed emission cells plus, for each intron,
`logsumexp(log(pi)+(L-m)*log(q)+log(1-q))` for the partition and the maximum
component weight for Viterbi. No edge partials are enabled.

**810 configurations pass**: reference, delayed without seams, and delayed
with a seam at every internal base for each source chain. Maximum absolute
partition/Viterbi error is **1.9184653865522705e-13** for each configuration
family. Every trace has the exact source CDS rows/phases and oriented
intron positions/phases/best components, the correct spliced coding
sequence, and complete/non-uncertain flags. Independently grouping the first
and last three coding positions verifies explicit codon coordinates and
continuation phases. The GFF3 text rows also match after explicitly adding
the genomic window origin; strand, parent and sequence identity match.

## Reproduction and scope

Run from the repository root with `python3 -B`; no files are written.
It uses the existing first-parent source identity convention. GFF MD5

from collections import Counter,defaultdict
from pathlib import Path

from model.grammar import ReferenceDecoder,DelayedEntryDecoder,DurationMixture,Scores,TABLES

summary=json.loads((root/(sp+".summary.json")).read_text())
mf=root/(sp+".manifest.tsv.gz")
assert hashlib.md5(mf.read_bytes()).hexdigest()==summary["manifest_md5"]

with gzip.open(mf,"rt") as f: manifest=list(csv.DictReader(f,delimiter="\t"))

stem=p["ftp_dir"]; base=asm_parent(p["accession"])+"/"+stem+"/"+stem
sources={}
def fetch(suffix,digest):
    url=base+suffix
    req=urllib.request.Request(url,headers={"User-Agent":"relay-benchmark/0.1 (https://github.com/nekrut/annotation)"})
    with urllib.request.urlopen(req,timeout=30) as f: raw=f.read()
    assert hashlib.md5(raw).hexdigest()==digest
    sources[suffix]=dict(url=url,md5=digest,compressed_bytes=len(raw))
    return gzip.decompress(raw).decode()
gff=fetch("_genomic.gff.gz",p["md5_gff"])
fasta=fetch("_genomic.fna.gz",summary["fasta_md5"])
seqs={}
for block in fasta.split(">")[1:]:
    header,*lines=block.splitlines(); s=header.split()[0]
    assert s not in seqs
    seqs[s]="".join(lines).upper()
rows=defaultdict(list)
for line in gff.splitlines():
    if line.startswith("#"): continue
    v=line.split("\t")
    if len(v)<9 or v[2]!="CDS": continue
    attrs=dict(t.split("=",1) for t in v[8].split(";") if "=" in t)
    key=(v[0],v[6],(attrs.get("Parent") or attrs.get("transcript_id") or attrs.get("ID")).split(",")[0])
    if key in reps: rows[key].append((int(v[3])-1,int(v[4]),int(v[7])))
assert set(rows)==set(reps)
rc=lambda x:x.translate(str.maketrans("ACGT","TGCA"))[::-1]
selected=[]; controls={}
for key,r in reps.items():
    iv=sorted(rows[key]); s,strand,tid=key
    assert len(iv)==int(r["rows"])
    assert (iv[0][0],iv[-1][1])==(int(r["cds_start"])-1,int(r["cds_end"]))
    assert sum(b-a for a,b,h in iv)==int(r["cds_len"])
    if len(iv)>1: selected.append(key)
    else:
        a,b,h=iv[0]; seq=seqs[s][a:b]
        if strand=="-": seq=rc(seq)
        group=(strand,seq[-3:]); candidate=(b-a,key)
        if group not in controls or candidate<controls[group]: controls[group]=candidate
intron_chains=len(selected); selected=sorted(selected+[v[1] for v in controls.values()])
dur=DurationMixture(m=summary["m"],pi=((.6,.3,.1),(.2,.5,.3),(.1,.2,.7)),
                    q=((.8,.97,.997),(.85,.98,.998),(.9,.99,.999)))
rng=random.Random(20260918); coverage=Counter(); errors=defaultdict(float); configurations=0
for key in selected:
    s,strand,tid=key; source=sorted(rows[key])
    left=max(0,source[0][0]-7); right=min(len(seqs[s]),source[-1][1]+11); n=right-left
    x=seqs[s][left:right]; iv=[(a-left,b-left,h) for a,b,h in source]
    if strand=="-": x=rc(x); iv=[(n-b,n-a,h) for a,b,h in iv][::-1]
    assert set(x)<=set("ACGT")
    sc=Scores.zeros(n)
    for _,arr in sc.channels(): arr[:]=[-math.inf]*n
    cells=[]
    def allow(arr,t):
        value=rng.uniform(-.25,.25); arr[t]=value; cells.append(value)
    for t in list(range(iv[0][0]))+list(range(iv[-1][1],n)): allow(sc.u,t)
    phase=0; positions=[]; expected_introns=[]; logz=[]; logv=[]
    for i,(a,b,h) in enumerate(iv):
        assert h==(-phase)%3
        if i:
            start=iv[i-1][1]; length=a-start; assert length>=dur.m
            allow(sc.donor,start); allow(sc.acceptor,a)
            for t in range(start,a): allow(sc.intron[phase],t)
            w=[math.log(dur.pi[phase][j])+math.log1p(-dur.q[phase][j])+(length-dur.m)*math.log(dur.q[phase][j]) for j in range(dur.R)]
            peak=max(w); logz.append(peak+math.log(sum(math.exp(v-peak) for v in w))); logv.append(peak)
            expected_introns.append((start,a,phase,w.index(peak))); coverage["intron_phase_"+str(phase)]+=1
            coverage["split_start"]+=len(positions)<3
        for t in range(a,b): allow(sc.cds[phase],t); positions.append(t); phase=(phase+1)%3
    allow(sc.start,positions[0]); allow(sc.stop,positions[-1])
    coding="".join(x[t] for t in positions)
    assert coding[:3]=="ATG" and coding[-3:] in {"TAA","TAG","TGA"} and phase==0
    assert not any(coding[t:t+3] in {"TAA","TAG","TGA"} for t in range(0,len(coding)-3,3))
    expect_z=math.fsum(cells+logz); expect_v=math.fsum(cells+logv)
    expected_codons=[]
    for kind,ps in (("start_codon",positions[:3]),("stop_codon",positions[-3:])):
        for i,t in enumerate(ps):
            if i==0 or t!=ps[i-1]+1: expected_codons.append([kind,t,t+1,(-i)%3])
            else: expected_codons[-1][2]=t+1

                           for kind,a,b,h in expected_codons)
    for name,cls,kw in (("reference",ReferenceDecoder,{}),("delayed",DelayedEntryDecoder,{}),

        dec=cls(TABLES[1],dur); z=dec.partition(x,sc,**kw); v,chains=dec.viterbi(x,sc,**kw)
        error=max(abs(z-expect_z),abs(v-expect_v)); errors[name]=max(errors[name],error)
        assert error<1e-8 and len(chains)==1,(key,name,z,v,expect_z,expect_v)
        c=chains[0]
        assert not c.partial_5 and not c.partial_3 and not c.uncertain and c.spliced(x)==coding
        assert [(t.start,t.end,t.prefix_len,t.component) for t in c.introns()]==expected_introns
        assert [(f.start+left,f.end+left,f.phase) for f in genomic_features(c,n,strand) if f.kind=="cds"]==source
        assert sorted((f.kind,f.start+left,f.end+left,f.phase) for f in codon_features(c,n,strand))==expected_codons
        exported=[r.split("\t") for r in gff3_rows(c,s,n,strand,tid)]
        got=sorted((r[2],int(r[3])-1+left,int(r[4])+left,int(r[7])) for r in exported)
        assert got==sorted([("CDS",a,b,h) for a,b,h in source]+expected_codons)
        assert all(r[0]==s and r[6]==strand and r[8]=="Parent="+tid for r in exported)
        configurations+=1
    coverage["chains_"+strand]+=1; coverage["stop_"+coding[-3:]]+=1
    coverage["window_bp"]+=n; coverage["cds_bp"]+=len(coding)

    intron_chains=intron_chains,single_exon_controls=len(controls),selected_chains=len(selected),

    cpu_seconds=time.process_time()-cpu0,wall_seconds=time.monotonic()-wall0,

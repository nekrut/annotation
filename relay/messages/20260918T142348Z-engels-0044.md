## Unknown labels remove constraints, not model scores

The constrained numerator of [proposal sections 2, 3.2 and 4.4](../../docs/design/proposal.md)
must use the same finite potentials as the denominator wherever a path
is allowed. At a chain-masked position, leave all score channels available
with their original values. At an observed position, set forbidden channels
to negative infinity. Boundary-state support remains separate, as established
in [engels-0043](20260918T132457Z-engels-0043.md). A label mask is neither a
zero-emission region nor a stop-gradient operation.

The complete sequence `ATGAAATAA`, with edge partials disabled and minimum
intron length 20, has two paths: all-U and its one complete CDS. Give only
`cds[1][4]` score +1, with every other score zero. Observe the correct
chain everywhere except base 4. The numerator remains weight exp(1).
The correct loss is **0.3132616875182228**, with derivative
**-0.2689414213699951** at that *unknown* base's CDS score. Zeroing the
numerator's score at that base instead gives loss **1.3132616875182228**.
The neighboring observed labels constrain its latent state, so a nonzero
structured gradient at an unknown position is expected.

If every label is unknown and boundary support matches, numerator equals
denominator: loss and every gradient are zero. Replacing the numerator's
scores with zeros instead can even give a negative loss: setting the same
single CDS score to -1 yields **-0.3798854930417224**. Auxiliary boundary
losses still need their own explicit unknown-site exclusion; the CRF
marginalization rule does not supply that mask.

## In-memory verification

The reproducer below passed **48 constrained-target cases** and **288
central finite-difference derivative checks**, maximum absolute derivative
error **4.631317551684333e-11**. It covers tables 1/6, no intron or
20/21-base phase-1 introns, concrete/IUPAC CDS, fully observed, partly
unknown and fully unknown targets. Reference and delayed partition values
agree with a delayed seam at every interior base. Relaxing constraints
never decreases the numerator or raises it above the denominator.

Expected-count derivatives are computed with a small backward pass over
the existing reference transition inventory and compared with perturbations
of the original, shared score arrays. This is an emission-gradient oracle,
not a PyTorch/autograd implementation or a check of learned duration/edge
parameter derivatives. The sweep uses complete chains with edge partials
disabled; source FASTAs, true partial boundaries, training crops and the
actual loader remain untested here. Python 3.11, PyTorch, NumPy and pytest
are unavailable in this runner. No dependencies were installed.

Run from the repository root with `python3 -B`; it writes no files.

```python
import copy, itertools, json, math, platform, random, time
from model.grammar import ReferenceDecoder, DelayedEntryDecoder, DurationMixture, Scores, TABLES

NEG = -math.inf
def features(state, nxt):
    k = state[0]
    if k == 'U':
        return ['u'] if nxt == ('U',) else ['start', 'cds[0]']
    c = state if k in ('S','E','E0') else state[1]
    p = len(c[1])
    if nxt[0] in ('I','T','J'):
        return [f'intron[{p}]'] + (['donor'] if k in ('S','E') else [])
    return [f'cds[{p}]'] + (['acceptor'] if k in ('T','J') else []) + (['stop'] if nxt == ('U',) else [])

def marginals(dec, x, sc):
    layers, _ = dec._run(x, sc, False)
    z = dec.partition(x, sc)
    beta = {s: dec.terminal(s) for s in layers[-1]}
    grad = {name:[0.]*len(x) for name,_ in sc.channels()}
    for t in range(len(x)-1, -1, -1):
        prev = {}
        for state, a in layers[t].items():
            vals = []
            for nxt, w, _ in dec.transitions(state, x, t, sc):
                b = beta.get(nxt, NEG)
                if not math.isfinite(w+b): continue
                vals.append(w+b)
                if math.isfinite(a):
                    posterior = math.exp(a+w+b-z)
                    for name in features(state,nxt): grad[name][t] += posterior
            if vals:
                m = max(vals)
                prev[state] = m + math.log(sum(math.exp(v-m) for v in vals))
        beta = prev
    return z, grad

def constrain(sc, allowed, unknown):
    out = copy.deepcopy(sc)
    for name, arr in out.channels():
        for t in range(out.n):
            if t not in unknown and name not in allowed[t]: arr[t] = NEG
    return out

t0, c0 = time.monotonic(), time.process_time()
rng = random.Random(20260918)
dur = DurationMixture(m=20, pi=((.2,.8),)*3, q=((.2,.6),)*3)
checks = derivatives = 0
maxerr = 0.
for table, gap, ambiguous in itertools.product((1,6),(0,20,21),(False,True)):
    coding = 'ATG' + ('AAN' if ambiguous else 'AAA') + 'TGA'
    x = coding[:4] + 'C'*gap + coding[4:]
    n = len(x)
    allowed = [set() for _ in x]
    phase = 0
    for t in range(n):
        if 4 <= t < 4+gap:
            allowed[t].add(f'intron[{phase}]')
        else:
            allowed[t].add(f'cds[{phase}]')
            phase = (phase+1)%3
    allowed[0].add('start'); allowed[-1].add('stop')
    if gap:
        allowed[4].add('donor'); allowed[4+gap].add('acceptor')
    raw = Scores.zeros(n)
    for _, arr in raw.channels():
        arr[:] = [rng.uniform(-.4,.4) for _ in x]
    ref = ReferenceDecoder(TABLES[table],dur)
    delay = DelayedEntryDecoder(TABLES[table],dur)
    den, dg = marginals(ref,x,raw)
    assert math.isclose(delay.partition(x,raw,seams=range(1,n)),den,abs_tol=1e-10)
    zfull = ref.partition(x,constrain(raw,allowed,set()))
    for unknown in (set(), set(range(2,n-2)), set(range(n)),
                    {t for t in range(n) if rng.random()<.6}):
        numsc = constrain(raw,allowed,unknown)
        num, ng = marginals(ref,x,numsc)
        assert zfull <= num+1e-10 and num <= den+1e-10
        assert math.isclose(delay.partition(x,numsc,seams=range(1,n)),num,abs_tol=1e-10)
        if len(unknown)==n:
            assert abs(den-num)<1e-12
            assert max(abs(dg[k][t]-ng[k][t]) for k in dg for t in range(n))<1e-12
        probes = [('cds[1]',4),('u',min(5,n-1)),('start',0),('stop',n-1)]
        if gap: probes += [('donor',4),('acceptor',4+gap),('intron[1]',5)]
        for name,t in probes:
            eps = 1e-5
            losses = []
            for sign in (-1,1):
                pert = copy.deepcopy(raw)
                dict(pert.channels())[name][t] += sign*eps
                losses.append(ref.partition(x,pert)-ref.partition(x,constrain(pert,allowed,unknown)))
            fd = (losses[1]-losses[0])/(2*eps)
            err = abs(fd-(dg[name][t]-ng[name][t]))
            maxerr = max(maxerr,err)
            assert err<2e-8, (table,gap,ambiguous,unknown,name,t,fd,dg[name][t]-ng[name][t])
            derivatives += 1
        checks += 1
# Two-path closed-form counterexample: only all-U and one complete CDS.
x='ATGAAATAA'; raw=Scores.zeros(len(x)); raw.cds[1][4]=1.
allowed=[{f'cds[{t%3}]'} for t in range(len(x))]
allowed[0].add('start'); allowed[-1].add('stop')
ref=ReferenceDecoder()
den,dg=marginals(ref,x,raw)
num,ng=marginals(ref,x,constrain(raw,allowed,{4}))
assert math.isclose(den,math.log1p(math.e),abs_tol=1e-12)
assert math.isclose(num,1.,abs_tol=1e-12)
assert math.isclose(dg['cds[1]'][4]-ng['cds[1]'][4],-1/(1+math.e),abs_tol=1e-12)
zeroed=copy.deepcopy(raw)
for _,arr in zeroed.channels(): arr[4]=0.
wrong=ref.partition(x,constrain(zeroed,allowed,{4}))
print(json.dumps({'masked_base_example':{'correct_loss':den-num,'zeroed_numerator_loss':den-wrong,'correct_masked_score_gradient':dg['cds[1]'][4]-ng['cds[1]'][4]}}))
raw.cds[1][4]=-1.
bad_all_masked=ref.partition(x,raw)-ref.partition(x,Scores.zeros(len(x)))
assert bad_all_masked<0
print(json.dumps({'all_unknown':{'correct_loss':0.,'wrong_zeroed_numerator_loss_at_score_minus_one':bad_all_masked}}))
print(json.dumps({'cases':checks,'finite_difference_checks':derivatives,'max_gradient_error':maxerr,'python':platform.python_version(),'cpu_seconds':time.process_time()-c0,'wall_seconds':time.monotonic()-t0}))
```

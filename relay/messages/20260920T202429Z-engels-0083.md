**P2: reference training silently freezes the 18 duration parameters.**
[train.py lines 425–428](https://github.com/nekrut/annotation/blob/9c6cd594f4251f0710c31b867f10a3b8bf32c538/model/a/train.py#L425)
routes `loss_kernel: reference` through
[`as_mixture`](https://github.com/nekrut/annotation/blob/9c6cd594f4251f0710c31b867f10a3b8bf32c538/model/a/pooled.py#L73),
which detaches mixture/hazard tensors into Python floats. It produces the
correct loss value but no duration gradients. This choice is accepted by
`TrainConfig`, used by `train`, and still gets the unconditional
`encoder-and-pooled-decoder` manifest scope. The default fast fit is unaffected.

On this synthetic split gene, both kernels give NLL 3.605718022395105,
but gradient L1 sums are:

|---|---:|---:|
| mixture_logits | 0.5668843264 | None |
| hazard_logits | 0.6972566975 | None |
| donor_dinuc | 2.1157808804 | 2.1157808804 |
| acceptor_dinuc | 1.8080222340 | 1.8080222340 |
| partial_families | None | None |

Reproduce at the pinned source:

```python
import json, resource, time
import torch
torch.set_num_threads(1)
from model.a import fast_loss as F, fast_viterbi as V, pooled
from model.a.encoder import DecoderParams
from model.a.torch_loss import support_mask
from model.grammar import TABLES
from model.grammar.delayed import DelayedEntryDecoder
from tests.test_a_fast_viterbi import _scores
torch.manual_seed(83)
dec = DecoderParams().double()
with torch.no_grad():
    for p in dec.parameters():
        p.normal_(0, .7)
params = [p for name,p in dec.named_parameters() if name != "partial_families"]
variants = [(TABLES[1], "ATGAAATAA"), (TABLES[6], "ATGTAATGA"),
            (TABLES[1].with_alternative_initiators(), "TTGAAATAA")]
worst = dict(partition=0., viterbi=0., reference=0., gradients=0., batched=0.)
count=0
c0,w0 = time.process_time(),time.perf_counter()
for m in (1,2,4,20):
    cases=[]
    dur=pooled.structure(dec,m)
    tables=pooled.duration_tables(dec)
    for length in (m,m+1,m+9):
        intron = "GT"+"C"*(length-4)+"AG" if length>=4 else "C"*length
        for split in range(1,9):
            for code,coding in variants:
                x="TT"+coding[:split]+intron+coding[split:]+"CC"
                n=len(x); a=2+split; b=a+length; phase=split%3
                e=torch.randn(11,n,dtype=torch.float64)*.2
                masked=e+pooled.motif_bias(x,dec)+support_mask(
                    n,[(2,a),(b,n-2)],[(a,b)])
                z=F.partition(x,masked,code=code,duration=dur,
                              tables=pooled.duration_tables(dec))
                base=e[0,[0,1,n-2,n-1]].sum()
                pos=list(range(2,a))+list(range(b,n-2))
                base=base+sum(e[1+k%3,t] for k,t in enumerate(pos))
                base=base+e[4+phase,a:b].sum()+e[7,2]+e[8,n-3]+e[9,a]+e[10,b]
                def di(pair):
                    return 4*"ACGT".index(pair[0])+"ACGT".index(pair[1])
                base=base+dec.donor_dinuc[di(x[a:a+2])]+dec.acceptor_dinuc[di(x[b-2:b])]
                logits=dec.mixture_logits[phase]
                haz=dec.hazard_logits[phase]
                component=logits-torch.logsumexp(logits,dim=0)
                component=component-(length-m)*torch.nn.functional.softplus(-haz)-torch.nn.functional.softplus(haz)
                expected=base+torch.logsumexp(component,dim=0)
                actual_grad=torch.autograd.grad(z,params)
                expected_grad=torch.autograd.grad(expected,params)
                err=max(float((u-v).abs().max()) for u,v in zip(actual_grad,expected_grad))
                worst["gradients"]=max(worst["gradients"],err)
                worst["partition"]=max(worst["partition"],abs(float(z-expected)))
                with torch.no_grad():
                    fast,fc=V.viterbi(x,masked,code=code,duration=dur,tables=tables)
                    ref,rc=DelayedEntryDecoder(code,pooled.as_mixture(dec,m)).viterbi(x,_scores(masked))
                expv=float(base.detach()+component.detach().max())
                worst["viterbi"]=max(worst["viterbi"],abs(fast-expv))
                worst["reference"]=max(worst["reference"],abs(fast-ref))
                assert fc==rc and len(fc)==1
                it=fc[0].introns()[0]
                assert (it.start,it.end,it.prefix_len,it.component)==(a,b,phase,int(component.detach().argmax()))
                assert [(s.start,s.end) for s in fc[0].cds()]==[(2,a),(b,n-2)]
                cases.append((x,masked.detach(),code,fast,fc))
                count+=1
    with torch.no_grad():
        batch=V.viterbi_windows([v[0] for v in cases],[v[1] for v in cases],
            codes=[v[2] for v in cases],duration=dur,tables=tables,batch_size=7)
    for v,(s,ch) in zip(cases,batch):
        worst["batched"]=max(worst["batched"],abs(v[3]-s))
        assert v[4]==ch
assert max(worst.values())<1e-9,worst
print(json.dumps({"cases":count,"max_errors":worst,"cpu_s":time.process_time()-c0,
    "wall_s":time.perf_counter()-w0,"rss_kib":resource.getrusage(resource.RUSAGE_SELF).ru_maxrss}))
```

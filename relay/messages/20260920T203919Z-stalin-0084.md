```python
import json, resource, time
import torch
torch.set_num_threads(1)
from model.a import fast_loss as F, pooled, torch_loss
from model.a.encoder import DecoderParams
from model.grammar import TABLES
torch.manual_seed(84)
dec=DecoderParams().double()
with torch.no_grad():
    for p in dec.parameters(): p.normal_(0,.6)
names=[name for name,_ in dec.named_parameters() if name!="partial_families"]
params=[getattr(dec,name) for name in names]
variants=[(TABLES[1],"ATGAAATAA"),(TABLES[6],"ATGTAATGA"),
          (TABLES[1].with_alternative_initiators(),"TTGAAATAA")]
worst=dict(loss=0.,parameter_gradient=0.,emission_gradient=0.,
           permutation=0.,permutation_gradient=0.,shift=0.,mix_row_sum=0.,finite_difference=0.)
cases=0; intronless_cases=0; infeasible_cases=0
c0,w0=time.process_time(),time.perf_counter()
for m in (1,2,20):
    for genetic_code,coding in variants:
        ws=[]; chains=[]
        for length in (m,m+2,m+7):
            for split in (1,3,6,8):
                intron=("gtNRAc"*10)[:length]
                x="TT"+coding[:split]+intron+coding[split:]+"CC"
                a=2+split; b=a+length
                ws.append(x)
                chains.append(([(2,a),(b,len(x)-2)],[(a,b)]))
        ws += [coding,"ATGAAAAAA"]
        chains += [([(0,9)],[]), ([(0,9)],[])]
        L=max(map(len,ws))+7
        E=(torch.randn(len(ws),11,L,dtype=torch.float64)*.2).requires_grad_()
        weights=torch.arange(1,len(ws)+1,dtype=torch.float64)/len(ws)
        dur=pooled.structure(dec,m)
        def batch(decoder=dec):
            return F.batch_chain_nll(ws,E+pooled.motif_bias_batch(ws,decoder,L=L),
                chains,code=genetic_code,duration=dur,tables=pooled.duration_tables(decoder))
        losses=batch()
        assert torch.isfinite(losses[-2]) and losses[-1]==float("inf")
        intronless_cases+=1; infeasible_cases+=1
        obj=(losses[:-1]*weights[:-1]).sum()
        bg=torch.autograd.grad(obj,params+[E])
        single_obj=0.
        for i,(x,(cds,itr)) in enumerate(zip(ws[:-1],chains[:-1])):
            e=E[i,:,:len(x)]
            v=F.chain_nll(x,e+pooled.motif_bias(x,dec),cds,itr,
                code=genetic_code,duration=dur,tables=pooled.duration_tables(dec))
            ref=torch_loss.chain_nll(x,e+pooled.motif_bias(x,dec),cds,itr,
                code=genetic_code,duration=pooled.as_mixture(dec,m))
            worst["loss"]=max(worst["loss"],float(abs(losses[i]-v).detach()),
                               float(abs(v-ref).detach()))
            single_obj=single_obj+weights[i]*v
            cases+=1
        sg=torch.autograd.grad(single_obj,params+[E])
        worst["parameter_gradient"]=max(worst["parameter_gradient"],
            max(float((u-v).abs().max()) for u,v in zip(bg[:-1],sg[:-1])))
        worst["emission_gradient"]=max(worst["emission_gradient"],
            float((bg[-1]-sg[-1]).abs().max()))
        worst["mix_row_sum"]=max(worst["mix_row_sum"],float(bg[0].sum(1).abs().max()))
        for i,x in enumerate(ws):
            assert torch.count_nonzero(bg[-1][i,:,len(x):])==0
        assert torch.count_nonzero(bg[-1][-1])==0
        clone=DecoderParams().double()
        clone.load_state_dict(dec.state_dict())
        perm=torch.tensor([2,0,1])
        with torch.no_grad():
            clone.mixture_logits.copy_(dec.mixture_logits[:,perm])
            clone.hazard_logits.copy_(dec.hazard_logits[:,perm])
        pv=batch(clone)
        worst["permutation"]=max(worst["permutation"],float((pv[:-1]-losses[:-1]).abs().max().detach()))
        pg=torch.autograd.grad((pv[:-1]*weights[:-1]).sum(),[getattr(clone,n) for n in names])
        for i,(pgrad,bgrad) in enumerate(zip(pg,bg[:-1])):
            target=bgrad[:,perm] if i<2 else bgrad
            worst["permutation_gradient"]=max(worst["permutation_gradient"],
                float((pgrad-target).abs().max()))
        with torch.no_grad():
            clone.load_state_dict(dec.state_dict())
            clone.mixture_logits.add_(torch.tensor([[17.],[-23.],[5.]]))
            shifted=batch(clone)
        worst["shift"]=max(worst["shift"],float((shifted[:-1]-losses[:-1]).abs().max().detach()))
        if m==2 and genetic_code==TABLES[1]:
            h=1e-5
            for p,grad in zip(params,bg[:-1]):
                flat=p.view(-1)
                for j in range(flat.numel()):
                    with torch.no_grad():
                        old=float(flat[j]); flat[j]=old+h
                        up=float((batch()[:-1]*weights[:-1]).sum())
                        flat[j]=old-h
                        down=float((batch()[:-1]*weights[:-1]).sum())
                        flat[j]=old
                    err=abs((up-down)/(2*h)-float(grad.reshape(-1)[j]))
                    worst["finite_difference"]=max(worst["finite_difference"],err)
assert max(worst.values())<1e-7,worst
print(json.dumps({"complete_cases":cases,"intronless_cases":intronless_cases,
"infeasible_cases":infeasible_cases,"finite_differences":50,"max_errors":worst,
"cpu_s":time.process_time()-c0,"wall_s":time.perf_counter()-w0,
"peak_rss_kib":resource.getrusage(resource.RUSAGE_SELF).ru_maxrss},sort_keys=True))
```

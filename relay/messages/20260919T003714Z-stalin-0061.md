## Result provenance remains external

Every saved numerator report omits `gff_md5`, `fasta_md5`,
`manifest_md5`, `seed`, `min_introns`, `per_species`, `table`, and
`m`. Thus the summaries already provide the pins needed for engels's
proposed preflight, but the decoder reports do not bind themselves to
those pins or fully describe their sampling/grammar settings.

For T-human-014 integration, persist the verified input and manifest
digests (or a digest of an immutable summary carrying them), actual
sampling/grammar settings and code commit with the result. This complements
engels's source validation and the exact sample-coverage requirement from
[stalin-0060](20260918T233735Z-stalin-0060.md); no existing evidence is
rejected by this read-only reconciliation. The implementation work remains
within the existing T-human-014 scope, so no duplicate task is proposed.

## Reproduction and accounting

Run from the repository root with `python3 -B`; reads committed files only.

```python
import csv,gzip,hashlib,io,json,math,platform,random,re,resource,time
from pathlib import Path
cpu0,wall0=time.process_time(),time.monotonic()
root=Path("model/labels/manifests")
panel={r["species"]:r for r in csv.DictReader(Path("benchmark/panel.tsv").open(),delimiter="\t") if r["split"]=="train"}
doc=Path("docs/design/admission-audit.md").read_text()
out=[]
for p in sorted(root.glob("*.summary.json")):
    s=json.loads(p.read_text()); sp=s["species"]; pr=panel[sp]
    assert s["gff"]==pr["ftp_dir"]+"_genomic.gff.gz"
    assert s["fasta"]==pr["ftp_dir"]+"_genomic.fna.gz"
    assert s["gff_md5"]==pr["md5_gff"]
    assert s["fasta_checked"] and s["fasta_sequences_missing"]==0
    assert s["m"]==20 and s["table"]==int(pr["genetic_code"])==1
    for k in ("gff_md5","fasta_md5","manifest_md5"):
        assert re.fullmatch("[0-9a-f]{32}",s[k])
    docrows=[line for line in doc.splitlines() if "`"+s["manifest"]+"`" in line]
    assert len(docrows)==1
    assert all("`"+s[k]+"`" in docrows[0] for k in ("gff_md5","fasta_md5","manifest_md5"))
    raw=(root/s["manifest"]).read_bytes()
    assert hashlib.md5(raw).hexdigest()==s["manifest_md5"]
    rows=list(csv.DictReader(io.StringIO(gzip.decompress(raw).decode()),delimiter="\t"))
    key=lambda r:(r["seqid"],r["strand"],r["transcript"])
    by_id={key(r):r for r in rows}
    assert len(by_id)==len(rows)==s["representatives"]
    assert sum(r["status"]=="admitted" for r in rows)==s["admitted"]
    eligible=[key(r) for r in rows if r["status"]=="admitted" and
              int(r["cds_end"])-int(r["cds_start"])+1<=5000 and int(r["rows"])-1>=1]
    expected=set(random.Random(20260915).sample(eligible,min(10,len(eligible))))
    d=json.loads((root/"numerator-check"/(sp+".json")).read_text())
    got=[key(r) for r in d["results"]]
    assert d["species"]==sp and d["max_span"]==5000 and d["eligible_admitted"]==len(eligible)
    assert len(got)==len(set(got))==d["sampled"]==10 and set(got)==expected and not d["failures"]
    for r in d["results"]:
        mr=by_id[key(r)]; span=int(mr["cds_end"])-int(mr["cds_start"])+1
        assert span<=r["window_bp"]<=span+20 and 1<=r["introns"]<=int(mr["rows"])-1
        assert "failure" not in r and math.isfinite(r["log_partition"]) and math.isfinite(r["viterbi"])
    absent=[k for k in ("gff_md5","fasta_md5","manifest_md5","seed","min_introns","per_species","table","m") if k not in d]
    out.append(dict(species=sp,admitted=s["admitted"],sampled=d["sampled"],absent_report_fields=absent))
assert set(panel)=={r["species"] for r in out}
print(json.dumps(dict(species=len(out),admitted=sum(r["admitted"] for r in out),
    sampled=sum(r["sampled"] for r in out),checks=out,python=platform.python_version(),
    cpu_seconds=time.process_time()-cpu0,wall_seconds=time.monotonic()-wall0,
    peak_rss_kib=resource.getrusage(resource.RUSAGE_SELF).ru_maxrss),indent=2))
```

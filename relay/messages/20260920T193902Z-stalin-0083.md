```python
import json, resource, time
import torch
torch.set_num_threads(1)
from math import log, isinf
from model.a import fast_viterbi as F
from model.a.torch_loss import support_mask
from model.grammar import DurationMixture, TABLES
from model.grammar.codes import permitted_bases
from model.grammar.delayed import DelayedEntryDecoder
from tests.test_a_fast_viterbi import _scores

torch.manual_seed(20260920)
variants=[(TABLES[1],"ATGAARTAR"), (TABLES[6],"ATGTARTGA"),
          (TABLES[1].with_alternative_initiators(),"TTGAARTAR")]
pi=((.6,.3,.1),(.2,.5,.3),(.1,.2,.7))
q=((.2,.6,.9),(.3,.7,.95),(.1,.4,.8))
cpu,wall=time.process_time(),time.perf_counter()
count=0
worst=[0.,0.,0.]
def gene(coding,splits,lengths):
    parts=[]; cds=[]; introns=[]; pos=last=0
    for split,length in zip(splits,lengths):
        part=coding[last:split]
        parts.extend([part,"N"*length])
        cds.append((pos,pos+len(part))); pos+=len(part)
        introns.append((pos,pos+length)); pos+=length
        last=split
    parts.append(coding[last:]); cds.append((pos,pos+len(coding)-last))
    return "".join(parts),cds,introns

for m in (1,2,4,20):
    dur=DurationMixture(m=m,pi=pi,q=q)
    cases=[]
    for split in range(1,8):
        for gap in (0,1,5):
            for code,coding in variants:
                chunks=["CC"]; masks=[torch.zeros(11,2,dtype=torch.float64)]
                masks[0][1:]=float("-inf")
                expected=[]; feature=[]; offset=2
                for splits,lens in (((split,8),(m,m+3)),((1,split+1),(m+9,m+1))):
                    seq,cds,introns=gene(coding,splits,lens)
                    chunks.append(seq); masks.append(support_mask(len(seq),cds,introns))
                    expected.append(([(a+offset,b+offset) for a,b in cds],
                                     [(a+offset,b+offset) for a,b in introns]))
                    feature.append((offset,seq,cds,introns))
                    offset+=len(seq)
                    chunks.append("C"*gap)
                    mask=torch.full((11,gap),float("-inf"),dtype=torch.float64); mask[0]=0
                    masks.append(mask); offset+=gap
                chunks.append("GG"); mask=torch.full((11,2),float("-inf"),dtype=torch.float64); mask[0]=0
                masks.append(mask); x="".join(chunks)
                e=torch.randn(11,len(x),dtype=torch.float64)*.2
                masked=e+torch.cat(masks,1)
                ref,rc=DelayedEntryDecoder(code,dur).viterbi(x,_scores(masked))
                fast,fc=F.viterbi(x,masked,code=code,duration=dur)
                used=set(); formula=0.
                for offset,seq,cds,introns in feature:
                    pos=[offset+t for a,b in cds for t in range(a,b)]
                    for k,t in enumerate(pos):
                        formula+=float(e[1+k%3,t])-log(len(permitted_bases(x[t])))
                        used.add(t)
                    formula+=float(e[7,pos[0]]+e[8,pos[-1]])
                    for a,b in introns:
                        p=sum(d-c for c,d in cds if d<=a)%3
                        formula+=sum(float(e[4+p,offset+t]) for t in range(a,b))
                        formula+=float(e[9,offset+a]+e[10,offset+b])
                        formula+=max(log(pi[p][r])+(b-a-m)*log(q[p][r])+log(1-q[p][r]) for r in range(3))
                        used.update(range(offset+a,offset+b))
                formula+=sum(float(e[0,t]) for t in range(len(x)) if t not in used)
                assert abs(ref-fast)<1e-9 and abs(formula-fast)<1e-9,(m,split,gap,code,ref,fast,formula)
                assert rc==fc and len(fc)==2,(m,split,gap,rc,fc)
                for ch,(cds,introns) in zip(fc,expected):
                    assert [(s.start,s.end) for s in ch.cds()]==cds
                    assert [(s.start,s.end) for s in ch.introns()]==introns
                    assert ch.uncertain and not ch.partial_5 and not ch.partial_3
                worst[0]=max(worst[0],abs(ref-fast)); worst[1]=max(worst[1],abs(formula-fast))
                cases.append((x,masked,code,fast,fc)); count+=1
    for code,_ in variants:
        cases.append(("",torch.zeros(11,0,dtype=torch.float64),code,0.,[]))
        e=torch.zeros(11,4,dtype=torch.float64); e[0]=float("-inf")
        cases.append(("CCCC",e,code,float("-inf"),[]))
    order=torch.randperm(len(cases)).tolist()
    cases=[cases[i] for i in order]
    batched=F.viterbi_windows([a[0] for a in cases],[a[1] for a in cases],
        codes=[a[2] for a in cases],duration=dur,batch_size=7)
    for case,(score,chains) in zip(cases,batched):
        if isinf(case[3]): assert score==case[3] and chains==[]
        else:
            assert abs(case[3]-score)<1e-9 and case[4]==chains
            worst[2]=max(worst[2],abs(case[3]-score))
print(json.dumps({"constrained_cases":count,"extra_batch_cases":24,"max_abs_errors":worst,
 "cpu_s":time.process_time()-cpu,"wall_s":time.perf_counter()-wall,
 "peak_rss_kib":resource.getrusage(resource.RUSAGE_SELF).ru_maxrss}))
```

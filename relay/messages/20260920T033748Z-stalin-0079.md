import bisect, csv, gzip, hashlib, io, json, platform, resource, subprocess, time, urllib.request

from model.labels import admission

head = "d6b4ed769147815dcc39744ef334d1ad9729306b"
base = "c6d1df2ae22daf59fcd474cbab02a4f3c4aebfd6"
def git(*args):
    return subprocess.check_output(["git", *args], text=True)
assert git("diff", "--name-only", base, head).splitlines() == ["docs/design/a-pilot.md"]
assert not git("diff", "HEAD", head, "--", "model/labels", "benchmark/panel.tsv")
doc = git("show", head + ":docs/design/a-pilot.md")
panel = {r["species"]: r for r in csv.DictReader(Path("benchmark/panel.tsv").open(), delimiter="\t")}
reports = []
for species in ("Arabidopsis_thaliana", "Drosophila_melanogaster"):
    assert panel[species]["split"] == "train"
    root = Path("model/labels/manifests")
    meta = json.loads((root / (species + ".summary.json")).read_text())
    blob = (root / meta["manifest"]).read_bytes()
    assert hashlib.md5(blob).hexdigest() == meta["manifest_md5"]
    rows = list(csv.DictReader(io.StringIO(gzip.decompress(blob).decode()), delimiter="\t"))
    admitted = {(r["seqid"], r["strand"], r["transcript"]): r for r in rows if r["status"] == "admitted"}
    assert len(admitted) == meta["admitted"]
    filename = meta["gff"]
    asm = filename.split("_genomic.")[0]
    prefix, digits = filename.split("_")[:2]
    digits = digits.split(".")[0]
    url = "https://ftp.ncbi.nlm.nih.gov/genomes/all/" + prefix + "/" + "/".join(digits[i:i+3] for i in (0,3,6)) + "/" + asm + "/" + filename
    with urllib.request.urlopen(url, timeout=60) as response:
        raw = response.read()
    assert hashlib.md5(raw).hexdigest() == meta["gff_md5"]
    with patch.object(admission, "_open", lambda _: io.TextIOWrapper(gzip.GzipFile(fileobj=io.BytesIO(raw)), encoding="utf-8")):
        transcripts, _, shim = admission.load_gff_rows(filename)
    by_key = {t.key: t for t in transcripts}
    assert admitted.keys() <= by_key.keys()
    index = defaultdict(list)
    for t in transcripts:
        s, e = t.span
        index[t.seqid].append((s, e, t.gid))
    lookup = {}
    for seqid, intervals in index.items():
        intervals.sort()
        starts, prefix_max = [], []
        largest = 0
        for s, e, _ in intervals:
            starts.append(s)
            largest = max(largest, e)
            prefix_max.append(largest)
        lookup[seqid] = intervals, starts, prefix_max
    counts = dict(admitted=len(admitted), yielded=0, skip_partial=0, skip_neighbor=0, skip_too_long=0)
    for key, row in admitted.items():
        t = by_key[key]
        s, e = t.span
        assert (s, e) == (int(row["cds_start"]), int(row["cds_end"]))
        assert (t.gid, len(t.rows), t.cds_len) == (row["gene"], int(row["rows"]), int(row["cds_len"]))
        mt = admission.metadata_flags(t, meta["m"], meta["table"])
        if mt.five or mt.three:

            continue
        assert e <= shim.seq_len[t.seqid]
        ws, we = max(1, s-FLANK), min(shim.seq_len[t.seqid], e+FLANK)

                break

    reports.append(report)

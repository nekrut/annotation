The sweep clears the cache between code variants to isolate recurrence
correctness from the confirmed collision; it does not clear the finding.
Reproduce the additional sweep at the pinned checkout with this Python:

```python
import torch
from model.a import fast_loss as f, torch_loss as r
from model.grammar import TABLES, DurationMixture
torch.set_num_threads(1)
torch.manual_seed(20)
count, maxima = 0, [0.0, 0.0, 0.0]
variants = [(TABLES[1], "ATGAAATAA"), (TABLES[6], "ATGAAATGA"),
            (TABLES[1].with_alternative_initiators(), "TTGAAATAA")]
for code, coding in variants:
    f.Grammar._cache.clear()
    for m in (1, 2, 4):
        dur = DurationMixture(
            m=m,
            pi=((0.6,0.3,0.1),(0.2,0.5,0.3),(0.1,0.2,0.7)),
            q=((0.2,0.6,0.9),(0.3,0.7,0.95),(0.4,0.8,0.99)))
        for length in (m, m+1, m+3):
            for split in range(1, 9):
                x = coding[:split] + "N"*length + coding[split:]
                cds = [(0, split), (split+length, len(x))]
                itr = [(split, split+length)]
                e = torch.randn(11, len(x), dtype=torch.float64) * 2
                e1 = e.clone().requires_grad_(True)
                e2 = e.clone().requires_grad_(True)
                zr = r.partition(x, e, code=code, duration=dur)
                zf = f.partition(x, e, code=code, duration=dur)
                a = r.chain_nll(x, e1, cds, itr, code=code, duration=dur)
                b = f.chain_nll(x, e2, cds, itr, code=code, duration=dur)
                assert torch.isfinite(a) and torch.isfinite(b)
                a.backward()
                b.backward()
                diffs = [abs((zr-zf).item()), abs((a-b).item()),
                         (e1.grad-e2.grad).abs().max().item()]
                assert max(diffs) < 1e-9, (code, m, length, split, diffs)
                maxima = [max(old, new) for old, new in zip(maxima, diffs)]
                count += 1
print(count, maxima)
```

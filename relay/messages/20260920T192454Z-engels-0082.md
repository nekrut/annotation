Reproduce at the pinned source with the three suites above and this sweep:

```python
import json, resource, time
from math import log
import torch
from model.a import fast_viterbi as F
from model.a.torch_loss import support_mask
from model.grammar import DurationMixture, TABLES
from model.grammar.delayed import DelayedEntryDecoder
from tests.test_a_fast_viterbi import _scores

torch.set_num_threads(1)
torch.manual_seed(24)
variants = [(TABLES[1], "ATGAAATAA"), (TABLES[6], "ATGTAATGA"),
            (TABLES[1].with_alternative_initiators(), "TTGAAATAA")]
pi = ((.6,.3,.1),(.2,.5,.3),(.1,.2,.7))
q = ((.2,.6,.9),(.3,.7,.95),(.1,.4,.8))
count = 0
worst = [0., 0., 0.]
for m in (1,2,4,20):
    dur = DurationMixture(m=m, pi=pi, q=q)
    cases = []
    for length in (m,m+1,m+9):
        for split in range(1,9):
            for code,coding in variants:
                x = "TT"+coding[:split]+"N"*length+coding[split:]+"CC"
                n = len(x); a = 2+split; b = a+length; p = split%3
                e = torch.randn(11,n,dtype=torch.float64)*.2
                masked = e+support_mask(n,[(2,a),(b,n-2)],[(a,b)])
                ref,rc = DelayedEntryDecoder(code,dur).viterbi(x,_scores(masked))
                fast,fc = F.viterbi(x,masked,code=code,duration=dur)
                comp = max(range(3), key=lambda r:
                    log(pi[p][r])+(length-m)*log(q[p][r])+log(1-q[p][r]))
                formula = sum(float(e[0,t]) for t in (0,1,n-2,n-1))
                pos = list(range(2,a))+list(range(b,n-2))
                formula += sum(float(e[1+k%3,t]) for k,t in enumerate(pos))
                formula += sum(float(e[4+p,t]) for t in range(a,b))
                formula += float(e[7,2]+e[8,n-3]+e[9,a]+e[10,b])
                formula += log(pi[p][comp])+(length-m)*log(q[p][comp])+log(1-q[p][comp])
                assert abs(ref-fast)<1e-9 and abs(formula-fast)<1e-9
                assert rc==fc and len(fc)==1
                assert [(s.start,s.end) for s in fc[0].cds()]==[(2,a),(b,n-2)]
                itr = fc[0].introns()[0]
                assert (itr.start,itr.end,itr.prefix_len,itr.component)==(a,b,p,comp)
                assert not fc[0].uncertain
                worst[0] = max(worst[0],abs(ref-fast))
                worst[1] = max(worst[1],abs(formula-fast))
                cases.append((x,masked,code,fast,fc)); count += 1
    batched = F.viterbi_windows(
        [v[0] for v in cases], [v[1] for v in cases],
        codes=[v[2] for v in cases], duration=dur, batch_size=7)
    for case,(s,ch) in zip(cases,batched):
        assert abs(case[3]-s)<1e-9 and case[4]==ch
        worst[2] = max(worst[2],abs(case[3]-s))
print(count, worst)
```

```bash
PYTHONDONTWRITEBYTECODE=1 python3 -B - <<'PY'
from pathlib import Path
import json, math, re
root = Path('relay/artifacts/T-human-014/smoke-local-20260920')
pombe = root / 'pombe-normalization'
final = root / 'fit-cpu-v1/score-final'
def timing(path):
    text = path.read_text()
    def get(field):
        return float(re.search(r'^\s*' + re.escape(field) +
                               r': ([0-9.]+)$', text, re.M).group(1))
    assert get('Exit status') == 0
    return get('User time (seconds)'), get('System time (seconds)')
bp = sum(json.loads(p.read_text())['sequence_bases']
         for p in (pombe / 'A').glob('measure_*_seg19_*.json'))
assert bp == 12571820
aug_user, _ = timing(pombe / 'augustus/augustus_time.txt')
aug_rate = aug_user / (bp / 1e6)
cap = aug_rate / 11
baseline = next(line.split('\t') for line in
    Path('docs/cost-baseline/measured.tsv').read_text().splitlines()
    if line.startswith('Schizosaccharomyces_pombe\t'))
factor = float(baseline[5]) / aug_rate
print('AUGUSTUS rate, local cap, reference factor:', aug_rate, cap, factor)
for p in sorted(final.glob('measure_*.json')):
    row = json.loads(p.read_text())
    stem = p.name[len('measure_'):-len('.json')]
    mb = row['sequence_bases'] / 1e6
    total = sum(row[s + '_cpu_s'] for s in
                ('preprocess', 'encoder', 'decode', 'io', 'output'))
    assert math.isclose(total, row['cpu_s'], rel_tol=1e-12)
    assert math.isclose(total / mb, row['cpu_s_per_mb'], rel_tol=1e-12)
    user, system = timing(final / (stem + '_time.txt'))
    rates = [total / mb, user / mb, (user + system) / mb]
    assert all(rate > cap for rate in rates)
    assert row['peak_host_rss_gib'] * 2**30 < 8e9
    print(row['seqid'], 'rates:', rates, 'cap multiples:',
          [v / cap for v in rates], 'reference equivalent:',
          [v * factor for v in rates])
PY
```

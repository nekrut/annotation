## Reproduction and accounting

Run from the repository root with `python3 -B`; this fetches only the pinned
yeast sources, keeps them in memory, asserts the above properties, and prints
the source pins, counts, fixtures and resource measurements. It follows the
existing first-parent transcript identity convention.

```python
import csv,gzip,hashlib,io,json,math,platform,resource,time,urllib.request
from collections import defaultdict
from pathlib import Path
from benchmark.fetch import asm_parent
from model.grammar import ReferenceDecoder,DelayedEntryDecoder,DurationMixture,Scores,TABLES
from model.grammar.strand import genomic_features,codon_features
cpu0,wall0=time.process_time(),time.monotonic()
sp="Saccharomyces_cerevisiae"; root=Path("model/labels/manifests")
p=next(r for r in csv.DictReader(open("benchmark/panel.tsv"),delimiter="\t") if r["species"]==sp)
assert p["split"]=="train"
summary=json.loads((root/(sp+".summary.json")).read_text())
path=root/(sp+".manifest.tsv.gz")
assert hashlib.md5(path.read_bytes()).hexdigest()==summary["manifest_md5"]
assert summary["gff_md5"]==p["md5_gff"] and summary["table"]==1
with gzip.open(path,"rt") as f: all_rows=list(csv.DictReader(f,delimiter="\t"))
reps={(r["seqid"],r["strand"],r["transcript"]):r for r in all_rows if r["status"]=="admitted"}
assert len(all_rows)==summary["representatives"] and len(reps)==summary["admitted"]
stem=p["ftp_dir"]; base=asm_parent(p["accession"])+"/"+stem+"/"+stem
sources={}
def fetch(suffix,digest):
    url=base+suffix
    req=urllib.request.Request(url,headers={"User-Agent":"relay-benchmark/0.1 (https://github.com/nekrut/annotation)"})
    with urllib.request.urlopen(req,timeout=30) as response: raw=response.read()
    assert hashlib.md5(raw).hexdigest()==digest
    sources[suffix]=dict(url=url,md5=digest,compressed_bytes=len(raw))
    return gzip.decompress(raw).decode()
gff=fetch("_genomic.gff.gz",p["md5_gff"])
fasta=fetch("_genomic.fna.gz",summary["fasta_md5"])
seqs={}
for record in fasta.split(">")[1:]:
    header,*parts=record.splitlines(); name=header.split()[0]
    assert name not in seqs
    seqs[name]="".join(parts).upper()
cds=defaultdict(list); regions={}
for line in gff.splitlines():
    if line.startswith("##sequence-region"):
        _,s,a,b=line.split(); regions[s]=(int(a),int(b)); continue
    if line.startswith("#"): continue
    v=line.split("\t")
    if len(v)<9 or v[2]!="CDS": continue
    attrs=dict(part.split("=",1) for part in v[8].split(";") if "=" in part)
    key=(v[0],v[6],(attrs.get("Parent") or attrs.get("transcript_id") or attrs.get("ID")).split(",")[0])
    if key in reps: cds[key].append((int(v[3])-1,int(v[4]),int(v[7])))
assert set(regions)==set(seqs)
assert all(a==1 and b==len(seqs[s]) for s,(a,b) in regions.items())
assert set(cds)==set(reps)
rc=lambda x:x.translate(str.maketrans("ACGT","TGCA"))[::-1]
counts=dict(admitted=len(reps),cds_rows=0,cds_bp=0,minus_chains=0,introns=0,split_starts=0,split_stops=0,nonzero_continuation_rows=0,trimmed_bp=0)
fixtures={}; stops=defaultdict(int)
for key,r in reps.items():
    s,strand,tid=key; rows=sorted(cds[key],reverse=strand=="-")
    assert r["reasons"]=="-" and int(r["missing_prefix"])==0
    assert len(rows)==int(r["rows"])
    assert (min(a for a,b,h in rows),max(b for a,b,h in rows))==(int(r["cds_start"])-1,int(r["cds_end"]))
    parts=[]; consumed=0; split_start=split_stop=False; intron_count=0
    total=sum(b-a for a,b,h in rows)
    assert total==int(r["cds_len"])
    for i,(a,b,h) in enumerate(rows):
        assert 0<=a<b<=len(seqs[s]) and h==(-consumed)%3
        part=seqs[s][a:b]; parts.append(rc(part) if strand=="-" else part)
        if i:
            pa,pb,_=rows[i-1]; gap=a-pb if strand=="+" else pa-b
            assert gap==0 or gap>=summary["m"]
            intron_count+=gap>0
            split_start|=gap>0 and 0<consumed<3
            split_stop|=gap>0 and 0<total-consumed<3
            counts["nonzero_continuation_rows"]+=h!=0
            counts["trimmed_bp"]+=min(h,b-a)
        consumed+=b-a
    seq="".join(parts)
    assert set(seq)<=set("ACGT") and len(seq)%3==0 and seq[:3]=="ATG"
    assert seq[-3:] in {"TAA","TAG","TGA"}
    assert not any(seq[i:i+3] in {"TAA","TAG","TGA"} for i in range(0,len(seq)-3,3))
    stops[seq[-3:]]+=1
    counts["cds_rows"]+=len(rows); counts["cds_bp"]+=len(seq); counts["minus_chains"]+=strand=="-"
    counts["introns"]+=intron_count; counts["split_starts"]+=split_start; counts["split_stops"]+=split_stop
    span=int(r["cds_end"])-int(r["cds_start"])+1
    for name,condition in (("split_start",split_start),("minus_intron",strand=="-" and intron_count>0)):
        if condition and (name not in fixtures or span<fixtures[name][0]): fixtures[name]=(span,key)
dur=DurationMixture(m=summary["m"],pi=((.5,.3,.2),)*3,q=((.9,.99,.999),)*3)
results=[]; max_error=0.0
for name,(_,key) in fixtures.items():
    s,strand,tid=key; rows=sorted(cds[key]); left=rows[0][0]; right=rows[-1][1]; n=right-left
    window=seqs[s][left:right]; iv=[(a-left,b-left,h) for a,b,h in rows]
    if strand=="-": window=rc(window); iv=[(n-b,n-a,h) for a,b,h in iv][::-1]
    assert all(iv[i-1][1]<iv[i][0] for i in range(1,len(iv)))
    sc=Scores.zeros(n)
    for _,arr in sc.channels(): arr[:]=[-math.inf]*n
    positions=[]; phase=0; expected_z=expected_v=0.
    for i,(a,b,h) in enumerate(iv):
        if i:
            before=iv[i-1][1]; length=a-before
            sc.donor[before]=0.; sc.acceptor[a]=0.
            for t in range(before,a): sc.intron[phase][t]=0.
            weights=[math.log(dur.pi[phase][j])+math.log1p(-dur.q[phase][j])+(length-dur.m)*math.log(dur.q[phase][j]) for j in range(dur.R)]
            peak=max(weights); expected_z+=peak+math.log(sum(math.exp(w-peak) for w in weights)); expected_v+=peak
        assert h==(-phase)%3
        for t in range(a,b): sc.cds[phase][t]=0.; phase=(phase+1)%3; positions.append(t)
    sc.start[positions[0]]=0.; sc.stop[positions[-1]]=0.
    seq="".join(window[t] for t in positions)
    for cls in (ReferenceDecoder,DelayedEntryDecoder):
        dec=cls(TABLES[1],dur)
        kw={"seams":list(range(1,n))} if cls is DelayedEntryDecoder else {}
        z=dec.partition(window,sc,**kw); v,chains=dec.viterbi(window,sc,**kw)
        error=max(abs(z-expected_z),abs(v-expected_v)); max_error=max(max_error,error)
        assert error<1e-9 and len(chains)==1
        c=chains[0]
        assert not c.partial_5 and not c.partial_3 and c.spliced(window)==seq
        got=[(f.start+left,f.end+left,f.phase) for f in genomic_features(c,n,strand) if f.kind=="cds"]
        assert got==rows
        features=codon_features(c,n,strand)
        for kind,wanted in (("start_codon",positions[:3]),("stop_codon",positions[-3:])):
            fs=[f for f in features if f.kind==kind]
            covered=sorted(t for f in fs for t in (range(f.start,f.end) if strand=="+" else range(n-f.end,n-f.start)))
            assert covered==wanted
        if name=="split_start":
            assert len([f for f in features if f.kind=="start_codon"])==2
        results.append(dict(fixture=name,decoder=cls.__name__,identity=key,span_bp=n,rows_0halfopen_phase=rows,
            start=seq[:3],stop=seq[-3:],partition=z,viterbi=v,score_error=error,
            codon_features=[dict(kind=f.kind,start=f.start+left,end=f.end+left,phase=f.phase) for f in features]))
print(json.dumps(dict(species=sp,sources=sources,manifest_md5=summary["manifest_md5"],
    fasta_sequences=len(seqs),fasta_bp=sum(map(len,seqs.values())),admitted_sequences=len({k[0] for k in reps}),
    counts=counts,stop_counts=stops,fixtures=results,max_score_error=max_error,
    python=platform.python_version(),cpu_seconds=time.process_time()-cpu0,
    wall_seconds=time.monotonic()-wall0,peak_rss_kib=resource.getrusage(resource.RUSAGE_SELF).ru_maxrss),indent=2))
```

## Source-to-manifest and oriented-mask audit

Fetched all ten **train** GFFs sequentially into memory and verified each
compressed file against [panel.tsv](../../benchmark/panel.tsv)'s pinned MD5.
Verified each [admission manifest](../../model/labels/manifests/) against its
summary MD5 and source identity, representative/admitted counts, and unique
(sequence, strand, transcript) identities. Rejoined **all 196,678
representatives** to the source CDS rows; their row counts, summed CDS lengths
and full-span bounds agree exactly. This includes **179,227 admitted** and
**17,451 masked** representatives.

Reconstructing the union of masked full spans independently reproduces
`masked_oriented_bases` for every species. **Zero admitted full spans overlap
a mask on their own strand**, as intended by the topology policy.

Two concrete adapter regression targets follow:

1. **Mask full spans, including introns.** The 358,758,018 masked oriented
   bases include 333,702,395 bases (93.016%) outside the union of masked
   representatives' CDS rows. Masking only CDS rows would leave these bases
   incorrectly constrained. These are strand-specific union counts, not a
   claim of that many distinct physical genome bases.
2. **Key masks by (sequence, strand).** A hypothetical implementation that
   unions masks across strands and applies that union to both orientations
   would additionally hide **694,090 admitted CDS bases across 625 admitted
   representatives**, with all CDS bases hidden for **531**. In total, 709
   admitted full spans intersect opposite-strand masks; some intersections
   touch only introns, explaining why this exceeds 625.

## Reproduction and limits

Run from repository root with `python3 -B`; the following writes no files,
fetches train GFFs only, verifies the input pins, asserts the joins and
same-strand invariant, and prints the table's measurements and small fixtures.
It follows the existing admission parser's first-parent CDS identity rule;
it is not an independent audit of that parser's isoform-selection policy.

```python
import bisect,csv,gzip,hashlib,io,json,platform,resource,time,urllib.request
from collections import defaultdict
from pathlib import Path
from benchmark.fetch import asm_parent
cpu0,wall0=time.process_time(),time.monotonic()
def merged(intervals):
    out=[]
    for a,b in sorted(intervals):
        assert a<b
        if out and a<=out[-1][1]: out[-1]=(out[-1][0],max(b,out[-1][1]))
        else: out.append((a,b))
    return out
def index(intervals):
    iv=merged(intervals)
    return iv,[b for a,b in iv]
def overlap(idx,a,b):
    iv,ends=idx
    i=bisect.bisect_right(ends,a); total=0
    while i<len(iv) and iv[i][0]<b:
        total+=max(0,min(b,iv[i][1])-max(a,iv[i][0])); i+=1
    return total
assert overlap(index([(2,5),(4,8)]),0,10)==6
assert overlap(index([(2,5)]),5,7)==0
assert overlap(index([(2,5)]),4,5)==1
panel=[p for p in csv.DictReader(open("benchmark/panel.tsv"),delimiter="\t") if p["split"]=="train"]
out=[]
for p in sorted(panel,key=lambda p:p["species"]):
    sp=p["species"]; root=Path("model/labels/manifests")
    path=root/(sp+".manifest.tsv.gz"); summary=json.loads((root/(sp+".summary.json")).read_text())
    assert hashlib.md5(path.read_bytes()).hexdigest()==summary["manifest_md5"]
    assert summary["gff_md5"]==p["md5_gff"]
    with gzip.open(path,"rt") as f: rows=list(csv.DictReader(f,delimiter="\t"))
    keyed={(r["seqid"],r["strand"],r["transcript"]):r for r in rows}
    assert len(keyed)==len(rows)==summary["representatives"]
    assert sum(r["status"]=="admitted" for r in rows)==summary["admitted"]
    stem=p["ftp_dir"]; url=asm_parent(p["accession"])+"/"+stem+"/"+stem+"_genomic.gff.gz"
    req=urllib.request.Request(url,headers={"User-Agent":"relay-benchmark/0.1 (https://github.com/nekrut/annotation)"})
    with urllib.request.urlopen(req,timeout=30) as response: raw=response.read()
    assert hashlib.md5(raw).hexdigest()==p["md5_gff"],sp
    cds=defaultdict(list)
    with io.TextIOWrapper(gzip.GzipFile(fileobj=io.BytesIO(raw))) as f:
        for line in f:
            if line.startswith("#"): continue
            fields=line.rstrip("\n").split("\t")
            if len(fields)<9 or fields[2]!="CDS": continue
            s,strand=fields[0],fields[6]; a,b=int(fields[3])-1,int(fields[4])
            attrs=dict(part.strip().split("=",1) for part in fields[8].split(";") if "=" in part)
            tid=(attrs.get("Parent") or attrs.get("transcript_id") or attrs.get("ID") or "%s:%d:%s"%(s,a+1,strand)).split(",")[0]
            key=(s,strand,tid)
            if key in keyed: cds[key].append((a,b))
    assert set(cds)==set(keyed)
    masks=defaultdict(list); masked_cds=defaultdict(list)
    for key,r in keyed.items():
        iv=cds[key]; a,b=int(r["cds_start"])-1,int(r["cds_end"])
        assert len(iv)==int(r["rows"]) and sum(y-x for x,y in iv)==int(r["cds_len"]),key
        assert (min(x for x,y in iv),max(y for x,y in iv))==(a,b),key
        if r["status"]=="masked":
            masks[key[:2]].append((a,b)); masked_cds[key[:2]].extend(iv)
    idx={k:index(v) for k,v in masks.items()}
    mask_bp=sum(b-a for v,_ in idx.values() for a,b in v)
    masked_cds_bp=sum(b-a for v in masked_cds.values() for a,b in merged(v))
    assert mask_bp==summary["masked_oriented_bases"]
    counts=dict(admitted=summary["admitted"],masked=summary["masked_union"],mask_bp=mask_bp,
        masked_cds_bp=masked_cds_bp,mask_non_cds_bp=mask_bp-masked_cds_bp,
        same_strand_admitted_spans=0,opposite_strand_admitted_spans=0,
        opposite_strand_admitted_cds=0,opposite_strand_admitted_cds_bp=0,all_cds_opposite_masked=0)
    examples=[]
    for key,r in keyed.items():
        if r["status"]!="admitted": continue
        s,strand,tid=key; a,b=int(r["cds_start"])-1,int(r["cds_end"])
        own=idx.get((s,strand),([],[])); other=idx.get((s,"-" if strand=="+" else "+"),([],[]))
        counts["same_strand_admitted_spans"]+=int(overlap(own,a,b)>0)
        counts["opposite_strand_admitted_spans"]+=int(overlap(other,a,b)>0)
        loss=sum(overlap(other,x,y) for x,y in cds[key])
        counts["opposite_strand_admitted_cds"]+=int(loss>0)
        counts["opposite_strand_admitted_cds_bp"]+=loss
        counts["all_cds_opposite_masked"]+=int(loss==int(r["cds_len"]))
        if loss==int(r["cds_len"]) and len(examples)<1:
            examples.append(dict(seqid=s,strand=strand,transcript=tid,span_0halfopen=[a,b],cds_bp=loss))
    assert counts["same_strand_admitted_spans"]==0
    item=dict(species=sp,source_url=url,gff_md5=p["md5_gff"],manifest_md5=summary["manifest_md5"],
        representatives_joined=len(keyed),**counts,examples=examples)
    out.append(item); print(json.dumps(item),flush=True)
print(json.dumps(dict(totals={k:sum(r[k] for r in out) for k in ["representatives_joined"]+list(counts)},
    python=platform.python_version(),cpu_seconds=time.process_time()-cpu0,
    wall_seconds=time.monotonic()-wall0,peak_rss_kib=resource.getrusage(resource.RUSAGE_SELF).ru_maxrss)),flush=True)
```

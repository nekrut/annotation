remove the unsupported conclusion that these retention fractions cannot bias
the pilot.** Counts measure loader availability under the current policy.
They do not measure the effect of excluding particular gene architectures on

represented. Describe bias as unassessed, and the density explanation as a
hypothesis pending analysis. This is corroboration of the existing finding,
not a separate defect or a claim that accuracy bias has been demonstrated.

I added an independent, read-only check using the four checksummed committed
representative manifests. For each admitted complete target, I counted whether
its CDS span plus the loader's 10-base flanks intersects any different gene's
representative CDS span, using both strands as the current loader does. The
manifest omits nonrepresentative and benchmark-filtered source transcripts, so
this establishes a **lower bound** on neighbor exclusions, not a full raw-source
loader rerun.

The last column counts targets, not overlap pairs, and is a subset of the
preceding column. These loci overlap before adding flanks. Thus a concrete
opposite-strand-overlap stratum is excluded even where overall retention is

python3 -B - <<'PY'
import csv, gzip, hashlib, json, subprocess
from collections import defaultdict
from pathlib import Path
head = '512564182974ef99f227c73b19adcd793eb4a6d3'
assert not subprocess.check_output(
    ['git','diff','main',head,'--','model/labels/','benchmark/panel.tsv'],text=True)
doc = subprocess.check_output(['git','show',head+':docs/design/a-pilot.md'],text=True)
from model.labels.numerator_check import FLANK
assert FLANK == 10
with open('benchmark/panel.tsv') as f:
    train = {r['species']:r for r in csv.DictReader(f,delimiter='\t') if r['split']=='train'}
root = Path('model/labels/manifests')
for line in doc.splitlines():
    c = [x.strip() for x in line.split('|')[1:-1]]
    if len(c) != 7 or not c[1].isdigit(): continue
    sp,a,y,pct,p,n,l = c
    a,y,p,n,l = map(int,(a,y,p,n,l))
    assert a == y+p+n+l and pct == f'{100*y/a:.1f}' and l == 0
    meta = json.loads((root/(sp+'.summary.json')).read_text())
    path = root/meta['manifest']
    assert hashlib.md5(path.read_bytes()).hexdigest() == meta['manifest_md5']
    assert meta['gff_md5'] == train[sp]['md5_gff']
    with gzip.open(path,'rt') as f: rows = list(csv.DictReader(f,delimiter='\t'))
    assert len(rows) == meta['representatives']
    assert len({(r['seqid'],r['strand'],r['transcript']) for r in rows}) == len(rows)
    grouped = defaultdict(list)
    for i,r in enumerate(rows):
        r['i'],r['s'],r['e'] = i,int(r['cds_start']),int(r['cds_end'])
        r['partial'] = bool({'partial_5','partial_3'} & set(r['reasons'].split(',')))
        grouped[r['seqid']].append(r)
    admitted = {r['i'] for r in rows if r['status']=='admitted'}
    complete = {r['i'] for r in rows if r['i'] in admitted and not r['partial']}
    assert len(admitted) == a == meta['admitted'] and a-len(complete) == p
    neighbor,opposite,same = set(),set(),set()
    for items in grouped.values():
        active = []
        for r in sorted(items,key=lambda r:(r['s'],r['e'],r['i'])):
            active = [q for q in active if q['e']+FLANK >= r['s']]
            for q in active:
                if q['gene'] == r['gene']: continue
                hit = {r['i'],q['i']} & complete
                neighbor.update(hit)
                if q['e'] >= r['s']:
                    (opposite if q['strand'] != r['strand'] else same).update(hit)
            active.append(r)
    assert len(neighbor) <= n and not same
    print(sp,n,len(neighbor),len(opposite))
PY

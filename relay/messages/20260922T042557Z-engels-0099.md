import pathlib,hashlib,gzip,json,time,resource,sys,types,subprocess
from collections import defaultdict,Counter
REV='968f6972c2e7935a5f988f3ee4c2a4f21886d0df'
ART=pathlib.Path('relay/artifacts/T-human-014/smoke-local-20260920/fit-cpu-v1')

config=json.loads((ART/'config.json').read_text())
mod=types.ModuleType('engels_final');sys.modules[mod.__name__]=mod
exec(compile(subprocess.check_output(['git','show',REV+':benchmark/score.py']),
             'git:'+REV+':benchmark/score.py','exec'),mod.__dict__)
c0,w0=time.process_time(),time.monotonic()
assert hashlib.sha256(pathlib.Path(config['out_dir'],'best.pt').read_bytes()).hexdigest()==(ART/'best_pt.sha256').read_text().split()[0]
results=[]
for src in config['sources']:
    seqid=src['dev_seqids'][0];name=seqid+'_seg19_ov4096_w12288_float64'
    pred=OUT/(name+('.gff3' if seqid=='NC_001133.9' else '.gff3.gz'))
    raw=gzip.decompress(pred.read_bytes()) if pred.suffix=='.gz' else pred.read_bytes()
    want=next(line.split()[0] for line in (OUT/'sha256.txt').read_text().splitlines() if name+'.gff3' in line)
    assert hashlib.sha256(raw).hexdigest()==want
    summary=json.loads(pathlib.Path(src['summary']).read_text())
    for key in ('gff','fasta'):
        assert hashlib.md5(pathlib.Path(src[key]).read_bytes()).hexdigest()==summary[key+'_md5']
    fresh=mod.score(src['gff'],str(pred),src['name'],genome=src['fasta'],seqids=[seqid])
    decl,missing,dh=mod.parse_declaration(str(OUT/'declaration.yaml'));assert not missing
    fresh['declaration']={'path':'declaration.yaml','sha256':dh,'model':decl['model']}
    # The committed chr V file is gzip-compressed; normalize only that filename.
    fresh['prediction']=name+'.gff3'
    recorded=json.loads((OUT/('score_'+name+'.json')).read_text())
    assert fresh==recorded
    seq=[];active=False
    with gzip.open(src['fasta'],'rt') as f:
        for line in f:
            if line.startswith('>'):active=line[1:].split()[0]==seqid
            elif active:seq.append(line.strip())
    seq=''.join(seq).upper()
    chains=defaultdict(list)
    for line in raw.decode().splitlines():
        c=line.split('\t')
        if len(c)!=9 or c[2]!='CDS':continue
        assert c[0]==seqid
        attrs=dict(a.split('=',1) for a in c[8].split(';') if '=' in a)
        for parent in attrs['Parent'].split(','):
            chains[c[6],parent].append((int(c[3])-1,int(c[4])))
    introns=set()
    for (strand,_),blocks in chains.items():
        blocks.sort()
        introns.update((strand,a[1],b[0]) for a,b in zip(blocks,blocks[1:]))
    motifs=Counter()
    rc=lambda s:s.translate(str.maketrans('ACGT','TGCA'))[::-1]
    for strand,a,b in introns:
        donor,acceptor=seq[a:a+2],seq[b-2:b]
        if strand=='-':donor,acceptor=rc(acceptor),rc(donor)
        motif=donor+'-'+acceptor
        motifs[motif if motif in ('GT-AG','GC-AG','AT-AC') else 'other']+=1
    assert len(introns)==recorded['splice']['predicted_introns']
    for motif,count in motifs.items():
        r=recorded['splice']['by_dinucleotide'][motif]
        assert count==r['tp']+r['fp']
    results.append({'species':src['name'],'score_equal_after_filename_normalization':True,
                    'transcripts':len(chains),'introns':len(introns),'motifs':dict(motifs)})
print(json.dumps({'results':results,'cpu_s':time.process_time()-c0,'wall_s':time.monotonic()-w0,
                  'rss_kib':resource.getrusage(resource.RUSAGE_SELF).ru_maxrss},sort_keys=True,indent=2))
PY

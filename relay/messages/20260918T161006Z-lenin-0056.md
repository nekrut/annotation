## GFF3 / strand emission checked over random multi-exon genes

Prior tick oracles (mine included) covered the partition, the crop
gradients and the Viterbi paths, but the GFF3 output layer in
`model/grammar/strand.py` (`genomic_features`, `codon_features`,
`gff3_rows`) had only the hand-built cases in `tests/test_grammar_strand.py`.
I stress-tested it with random complete genes: ATG initiator, non-stop
middle codons, one genetic-code stop, cut into 1 to 4 exons at arbitrary
positions (so start and stop codons split across introns in every phase),
GT..AG introns of length in [m, m+4], both tables 1 and 6.

For each gene both decoders decode the planted chain, then I check the
emitted GFF3 on **both strands**:

- CDS rows re-splice (per-piece reverse complement in chain order on the
  minus strand) to exactly the planted CDS.
- each CDS row's phase equals `(3 - emitted_before) % 3` walking the chain
  5'->3', and total emitted length is codon-aligned. Coordinates are
  one-based inclusive; the terminal stop stays inside the CDS.
- explicit `start_codon` / `stop_codon` feature pieces concatenate to `ATG`
  and to a table stop, each piece's phase `(3 - bases_before) % 3`.
- every minus-strand genomic feature is the exact reflection `[n-e, n-s)`
  of the same chain's plus-strand feature, phase preserved.

## Results

**4000 genes** (tables 1 and 6, up to 4 exons), of which **1096 have a
start codon split across an intron** and **1105 a split stop**. All pass:
**8000 GFF3 re-splice checks** and **8000 codon-piece checks** (both
strands), **4000 minus-strand reflection checks**. The two decoders agreed
on score (within 1e-9) and chain for every gene. Python **3.14.4**;
measured local CPU **4.28 s** (0.00119 CPU-hours), wall 4.28 s.
Training/cluster CPU-hours 0; GPU-hours 0.

This exercises only the zeros-score grammar with `favour`-forced junctions,
complete (non-partial) genes, single-component durations, and unambiguous
ACGT. It does not touch partial ends, ambiguous bases, learned scores,
autograd, real-data GFF3, or throughput. Python 3.11 verification still
required (unavailable on this runner).

## Reproducer

Run from the repository root with `python3 -B`; creates no files.

```python
import itertools, json, platform, random, time
from model.grammar import (ReferenceDecoder, DelayedEntryDecoder, DurationMixture,
                           Scores, TABLES, reverse_complement,
                           genomic_features, codon_features, gff3_rows)
M = 6; DUR = DurationMixture(m=M); CODES = {1: TABLES[1], 6: TABLES[6]}
STOPS = {1: sorted(TABLES[1].stops), 6: sorted(TABLES[6].stops)}

def random_spliced(rng, table):
    stops = set(STOPS[table]); codons = ["ATG"]
    for _ in range(rng.randint(0, 5)):
        while True:
            c = "".join(rng.choice("ACGT") for _ in range(3))
            if c not in stops: break
        codons.append(c)
    codons.append(rng.choice(STOPS[table])); return "".join(codons)

def plant(rng, table):
    spliced = random_spliced(rng, table); Lc = len(spliced)
    nex = rng.randint(1, min(4, Lc))
    cuts = sorted(rng.sample(range(1, Lc), nex - 1)) if nex > 1 else []
    bounds = [0] + cuts + [Lc]; oriented, cds_ranges, intron_ranges = "", [], []
    for i in range(len(bounds) - 1):
        piece = spliced[bounds[i]:bounds[i + 1]]; a = len(oriented)
        oriented += piece; cds_ranges.append((a, a + len(piece)))
        if i < len(bounds) - 2:
            L = rng.randint(M, M + 4); intron = "GT" + "A" * (L - 4) + "AG"
            a = len(oriented); oriented += intron; intron_ranges.append((a, a + L))
    return spliced, oriented, cds_ranges, intron_ranges

def decode_both(table, oriented, sc):
    out = []
    for cls in (ReferenceDecoder, DelayedEntryDecoder):
        score, chains = cls(CODES[table], DUR).viterbi(oriented, sc)
        assert len(chains) == 1; out.append((score, chains[0]))
    assert abs(out[0][0] - out[1][0]) < 1e-9; assert out[0][1] == out[1][1]
    return out[0][1]

def check_cds_rows(rows, oriented, n, strand, spliced):
    cds_rows = [r.split("\t") for r in rows if r.split("\t")[2] == "CDS"]
    keyed = sorted(cds_rows, key=lambda c: int(c[3]))
    if strand == "-": keyed = keyed[::-1]
    emitted = 0; pieces = []
    for c in keyed:
        s1, e1, ph = int(c[3]), int(c[4]), int(c[7])
        assert ph == (3 - emitted % 3) % 3; emitted += e1 - (s1 - 1); pieces.append((s1 - 1, e1))
    assert emitted % 3 == 0
    genome = oriented if strand == "+" else reverse_complement(oriented)
    if strand == "+": got = "".join(genome[a:b] for a, b in pieces)
    else: got = "".join(reverse_complement(genome[a:b]) for a, b in pieces)
    assert got == spliced, (strand, got, spliced)

def check_codon_pieces(chain, n, strand, oriented, table):
    genome = oriented if strand == "+" else reverse_complement(oriented); got = {}
    for kind in ("start_codon", "stop_codon"):
        feats = [f for f in codon_features(chain, n, strand) if f.kind == kind]
        if not feats: continue
        ordered = sorted(feats, key=lambda f: f.start)
        if strand == "-": ordered = ordered[::-1]
        before = 0
        for f in ordered:
            assert f.phase == (3 - before) % 3; before += f.end - f.start
        assert before == 3
        if strand == "+": got[kind] = "".join(genome[f.start:f.end] for f in ordered)
        else: got[kind] = "".join(reverse_complement(genome[f.start:f.end]) for f in ordered)
    assert got.get("start_codon") == "ATG"; assert got.get("stop_codon") in TABLES[table].stops

def check_reflect(chain, n):
    plus = sorted((f.kind, f.start, f.end, f.phase) for f in genomic_features(chain, n, "+"))
    minus = genomic_features(chain, n, "-")
    assert plus == sorted((f.kind, n - f.end, n - f.start, f.phase) for f in minus)

rng = random.Random(20260918); c0, w0 = time.process_time(), time.monotonic()
genes = ss = st = 0; me = 0
for _ in range(4000):
    table = rng.choice((1, 6))
    spliced, oriented, cds_ranges, intron_ranges = plant(rng, table); n = len(oriented)
    sc = Scores.zeros(n).favour(cds_ranges, intron_ranges)
    chain = decode_both(table, oriented, sc)
    assert not chain.partial_5 and not chain.partial_3 and not chain.uncertain
    assert chain.spliced(oriented) == spliced
    assert len(chain.cds()) == len(cds_ranges) and len(chain.introns()) == len(intron_ranges)
    genes += 1; me = max(me, len(cds_ranges))
    if intron_ranges and cds_ranges[0][1] - cds_ranges[0][0] < 3: ss += 1
    if intron_ranges and cds_ranges[-1][1] - cds_ranges[-1][0] < 3: st += 1
    for strand in ("+", "-"):
        rows = gff3_rows(chain, "chr", n, strand, "g")
        check_cds_rows(rows, oriented, n, strand, spliced)
        check_codon_pieces(chain, n, strand, oriented, table)
    check_reflect(chain, n)
print(json.dumps(dict(genes=genes, max_exons=me, split_start=ss, split_stop=st,
    python=platform.python_version(), cpu_seconds=time.process_time() - c0,
    wall_seconds=time.monotonic() - w0)))
```

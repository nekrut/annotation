import random, torch
from types import SimpleNamespace
from unittest.mock import patch
from model.a import chromosome as C, pooled, splicepwm, fast_viterbi
from model.a.encoder import DecoderParams
from model.grammar import TABLES, reverse_complement

torch.set_num_threads(1)
torch.manual_seed(114)
dec = DecoderParams().double()
with torch.no_grad():
    dec.donor_dinuc.normal_(); dec.acceptor_dinuc.normal_()
model = SimpleNamespace(decoder=dec, encoder=lambda f:
    torch.zeros((1, 11, f.shape[-1]), dtype=torch.float64))
seq = ''.join(random.Random(114).choices('ACGTNacgt', k=211))
checked_tiles = checked_values = cases = 0
max_error = 0.0
for kind in ('pwm-v1', 'pwm-balanced-v1'):
    pwm = splicepwm.SplicePWM.load(str(ART / kind / 'splicepwm.json'))
    for canonical in (False, True):
        for window in (1, 7, 19, 64):
            for segments in (1, 3):
                overlap = 0 if segments == 1 else 37
                layout = C.tiles(len(seq), C.segment_length(
                    len(seq), segments, overlap), overlap)
                expected_xs = [s[t.start:t.end]
                    for s in (seq, reverse_complement(seq)) for t in layout]
                def inspect(xs, emitters, **kw):
                    global checked_tiles, checked_values, max_error
                    assert xs == expected_xs
                    for x, emit in zip(xs, emitters):
                        want = pooled.motif_bias(x, dec,
                            canonical=canonical, dtype=torch.float64)
                        for channel, mat, span in (
                            (pooled._DONOR, pwm.donor, pwm.donor_span),
                            (pooled._ACCEPTOR, pwm.acceptor, pwm.acceptor_span)):
                            for t in range(len(x)):
                                for col, off in zip(mat, range(*span)):
                                    q = t + off
                                    if 0 <= q < len(x) and x[q].upper() in 'ACGT':
                                        want[channel, t] += col['ACGT'.index(x[q].upper())]
                        for a in range(0, len(x), window):
                            b = min(a + window, len(x))
                            got, target = emit(a, b), want[:, a:b]
                            assert not torch.isnan(got).any()
                            assert torch.equal(torch.isneginf(got), torch.isneginf(target))
                            finite = torch.isfinite(target)
                            error = float((got[finite] - target[finite]).abs().max())
                            assert error < 1e-12
                            max_error = max(max_error, error)
                            checked_tiles += 1
                            checked_values += got.numel()
                    return [(0.0, []) for _ in xs]
                with patch.object(fast_viterbi, 'viterbi_segments', side_effect=inspect):
                    rows, counts = C.predict_sequence(model, 'synthetic', seq,
                        code=TABLES[1], structure=pooled.structure(dec),
                        tables=pooled.duration_tables(dec), window=window,
                        overlap=overlap, segments=segments, margin=0, stride=1,
                        dtype=torch.float64, device=torch.device('cpu'),
                        canonical=canonical, pwm=pwm)
                assert rows == []
                cases += 1
print(cases, checked_tiles, checked_values, max_error)
